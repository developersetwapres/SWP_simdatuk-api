// ACCESS CONTROL LIST

Table users {
  id bigint [pk]
  role_id bigint [ref: < roles.id]
  name varchar(160) [not null]
  email varchar(160) [not null, unique]
  username varchar(160) [not null, unique]
  password varchar(255) [not null]
  nip varchar(13) [not null]
  nrp varchar(13) [not null]
  status boolean [default: true, note: '0=deactive, 1=active']
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`, note: 'ON UPDATE CURRENT_TIMESTAMP']
}

Table roles {
  id bigint [pk]
  name varchar(160) [not null, unique]
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`, note: 'ON UPDATE CURRENT_TIMESTAMP']
}

Table permissions {
  id bigint [pk]
  name varchar(160) [not null, unique]
  group varchar(160) [not null]
  sort int(13) [not null]
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`, note: 'ON UPDATE CURRENT_TIMESTAMP']
}

Table role_permissions {
  id bigint [pk]
  role_id bigint [ref: < roles.id]
  permission_id bigint [ref: < permissions.id]
  created_at timestamp [default: `CURRENT_TIMESTAMP`]
  updated_at timestamp [default: `CURRENT_TIMESTAMP`, note: 'ON UPDATE CURRENT_TIMESTAMP']
}
